const {chromium}=require(process.env.CODEX_PRIMARY_RUNTIME_NODE_MODULES+'/playwright');
const fs=require('fs'),http=require('http'),path=require('path');
(async()=>{
const base='/tmp/ta-site-remaining',out='/workspace/scratch/d1b15731b516/remaining-validation';
const server=http.createServer((req,res)=>{let p=path.join(base,new URL(req.url,'http://localhost').pathname);if(fs.existsSync(p)&&fs.statSync(p).isDirectory())p=path.join(p,'index.html');if(!fs.existsSync(p)){res.writeHead(404);res.end();return;}res.setHeader('Content-Type',({'.html':'text/html','.css':'text/css','.js':'application/javascript'})[path.extname(p)]||'application/octet-stream');fs.createReadStream(p).pipe(res);});
await new Promise(r=>server.listen(8766,'127.0.0.1',r));
const browser=await chromium.launch({executablePath:'/tmp/ta-render-toolchain/chrome-headless-shell-linux64/chrome-headless-shell',headless:true,args:['--no-sandbox']});const results=[];
try{for(const size of [{name:'desktop',width:1280,height:900},{name:'mobile',width:390,height:844}]){const ctx=await browser.newContext({viewport:size,javaScriptEnabled:false});await ctx.route('**/*',r=>r.request().url().startsWith('http://127.0.0.1:8766/')?r.continue():r.abort());const page=await ctx.newPage();for(const [route,anchor] of [['/agent-verify-simple/','external-review'],['/research/research-positioning/','open-questions-20260923'],['/research/research-positioning/zh.html','open-questions-20260923'],['/interpretation-verification-migration/','further-work-proposed-per-target-results']]){const r=await page.goto('http://127.0.0.1:8766'+route,{waitUntil:'load'});if(r.status()!==200)throw Error(route+' '+r.status());await page.locator('#'+anchor).scrollIntoViewIfNeeded();const d=await page.evaluate(()=>({width:innerWidth,scroll:document.documentElement.scrollWidth}));if(d.scroll>d.width+1)throw Error('overflow '+route+' '+JSON.stringify(d));const name=route.replace(/[^a-z0-9]/gi,'_')+size.name;await page.screenshot({path:out+'/'+name+'.jpg',type:'jpeg',quality:75});const broken=await page.locator('main a[href^="#"]').evaluateAll(as=>as.filter(a=>!document.getElementById(decodeURIComponent(a.getAttribute('href').slice(1)))).map(a=>a.getAttribute('href')));if(broken.length)throw Error('broken anchors '+route+' '+broken.join(','));results.push({route,viewport:size.name,status:200,javascript:false,no_page_overflow:true,local_anchors_valid:true});}await ctx.close();}fs.writeFileSync(out+'/render-check.json',JSON.stringify(results,null,2)+'\n');console.log(JSON.stringify(results));}finally{await browser.close();server.close();}
})().catch(e=>{console.error(e);process.exitCode=1});
