expected=version['files']; total=0; started=time.monotonic(); rows=[]
class OnlyZenodo(urllib.request.BaseHandler):
 def http_request(self,request):
  if urllib.parse.urlsplit(request.full_url).netloc != 'zenodo.org':raise RuntimeError('Host blocked by drill')
  if request.get_method()!='GET':raise RuntimeError('Only anonymous GET allowed')
  return request
 https_request=http_request
opener=urllib.request.build_opener(OnlyZenodo())
def fetch(url,label,*args,**kwargs):
 global total
 if time.monotonic()-started>600:raise RuntimeError('600-second acquisition budget')
 name=urllib.parse.unquote(urllib.parse.urlsplit(url).path.split('/')[-2])
 cap=expected[name]['bytes'] if name in expected else 2000000
 with opener.open(urllib.request.Request(url,headers={'Accept-Encoding':'identity'}),timeout=60) as r:
  if r.status!=200:raise RuntimeError('Non-200 acquisition')
  data=r.read(cap+1)
 if len(data)>cap:raise RuntimeError('Per-file budget exceeded')
 total+=len(data)
 if total>60_000_000:raise RuntimeError('60 MB budget exceeded')
 sha=hashlib.sha256(data).hexdigest()
 if name in expected:
  assert len(data)==expected[name]['bytes'] and sha==expected[name]['sha256']
 rows.append({'url':url,'bytes':len(data),'sha256':sha,'acquired_at_utc':datetime.now(timezone.utc).isoformat()})
 (root/'acquisition.json').write_text(json.dumps({'rows':rows,'total_bytes':total,'status':'partial'},indent=2)+'\n')
 print(label,len(data),flush=True)
 return data
